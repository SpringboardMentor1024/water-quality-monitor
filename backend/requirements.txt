# Web Framework
fastapi                  # The main framework to build the API
uvicorn[standard]        # The server that runs the FastAPI app

# Database
sqlalchemy               # ORM to interact with the database using Python classes
psycopg2-binary          # The driver that connects SQLAlchemy to PostgreSQL

# Data Validation
pydantic                 # Validates the data sent to the API
pydantic[email]          # Adds email format validation
email-validator          # Helper for pydantic to check if emails are valid

# Security & Authentication
passlib[bcrypt]          # Used to securely hash passwords
python-jose[cryptography] # Used to generate and verify JWT tokens (for login)
python-multipart         # Required to handle form data (like file uploads or login forms)

# Configuration
python-dotenv            # Loads secret variables (like DB password) from the .env file
